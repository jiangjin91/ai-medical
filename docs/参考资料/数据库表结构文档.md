## 字段约定

1. 主键统一使用 `BIGINT UNSIGNED AUTO_INCREMENT`。
2. 业务编号使用可读字符串，例如 `patient_no`、`visit_no`、`task_no`。
3. 状态字段统一使用 `VARCHAR(32)`，枚举值由业务服务约束。
4. 涉及患者隐私的字段只保存脱敏值或外部系统引用，不在本库保存完整身份证号、手机号、病历或报告正文。
5. 所有业务表保留 `created_at`、`updated_at`，必要时增加 `created_by`、`updated_by`。
6. MySQL 字符集统一使用 `utf8mb4`。

## 建表 SQL 草案

```sql
CREATE DATABASE IF NOT EXISTS `medical`
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_unicode_ci;

USE `medical`;

CREATE TABLE IF NOT EXISTS `medical_institutions` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键',
  `institution_code` VARCHAR(64) NOT NULL COMMENT '机构编码',
  `institution_name` VARCHAR(128) NOT NULL COMMENT '机构名称',
  `institution_type` VARCHAR(32) NOT NULL COMMENT '机构类型：hospital/campus/internet_hospital',
  `parent_id` BIGINT UNSIGNED DEFAULT NULL COMMENT '上级机构 ID，用于院区层级',
  `status` VARCHAR(32) NOT NULL DEFAULT 'active' COMMENT '状态',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_institution_code` (`institution_code`),
  KEY `idx_institution_parent` (`parent_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='医疗机构表';

CREATE TABLE IF NOT EXISTS `departments` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键',
  `institution_id` BIGINT UNSIGNED NOT NULL COMMENT '所属机构 ID',
  `department_code` VARCHAR(64) NOT NULL COMMENT '科室编码',
  `department_name` VARCHAR(128) NOT NULL COMMENT '科室名称',
  `department_type` VARCHAR(32) NOT NULL COMMENT '科室类型：outpatient/exam/lab/pharmacy/service',
  `location_desc` VARCHAR(255) DEFAULT NULL COMMENT '位置说明',
  `service_hours` VARCHAR(255) DEFAULT NULL COMMENT '服务时间说明',
  `status` VARCHAR(32) NOT NULL DEFAULT 'active' COMMENT '状态',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_department_code` (`institution_id`, `department_code`),
  KEY `idx_department_institution` (`institution_id`),
  CONSTRAINT `fk_department_institution` FOREIGN KEY (`institution_id`) REFERENCES `medical_institutions` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='科室表';

CREATE TABLE IF NOT EXISTS `patient_accounts` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键',

  `last_login_at` DATETIME DEFAULT NULL COMMENT '最近登录时间',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_account_no` (`account_no`),
  KEY `idx_account_channel` (`login_channel`, `status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='患者账号表';

  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键',

  `status` VARCHAR(32) NOT NULL DEFAULT 'active' COMMENT '就诊人状态',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_patient_no` (`institution_id`, `patient_no`),
  KEY `idx_patient_institution` (`institution_id`),
  CONSTRAINT `fk_patient_institution` FOREIGN KEY (`institution_id`) REFERENCES `medical_institutions` (`id`)

  CONSTRAINT `fk_auth_account` FOREIGN KEY (`account_id`) REFERENCES `patient_accounts` (`id`),
  CONSTRAINT `fk_auth_patient` FOREIGN KEY (`patient_id`) REFERENCES `patients` (`id`)
```
